cmake_minimum_required(VERSION 3.16)
project(OpenKeyChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
option(OPENKEY_SANITIZERS "Enable address and undefined-behavior checks" OFF)
option(OPENKEY_COVERAGE "Enable LLVM source coverage" OFF)
if(OPENKEY_COVERAGE)
    add_compile_options(-fprofile-instr-generate -fcoverage-mapping)
    add_link_options(-fprofile-instr-generate)
endif()
if(OPENKEY_SANITIZERS AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(MSVC)
    add_compile_options(/UNDEBUG /utf-8)
else()
    add_compile_options(-UNDEBUG)
endif()
if(WIN32)
    add_compile_definitions(NOMINMAX UNICODE _UNICODE)
endif()
set(ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../Sources/OpenKey/engine")
add_library(openkey_engine STATIC
    "${ENGINE}/Engine.cpp" "${ENGINE}/Vietnamese.cpp" "${ENGINE}/Macro.cpp"
    "${ENGINE}/SmartSwitchKey.cpp" "${ENGINE}/ConvertTool.cpp")
target_include_directories(openkey_engine PUBLIC "${ENGINE}")
find_package(Threads REQUIRED)
add_executable(engine_tests EngineTests.cpp EngineGlobals.cpp)
target_link_libraries(engine_tests PRIVATE openkey_engine Threads::Threads)
add_test(NAME engine COMMAND engine_tests)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SENDERS "${CMAKE_CURRENT_BINARY_DIR}/Senders.inc")
set(MAC_SENDER "${ENGINE}/../macOS/ModernKey/OpenKey.mm")
set(WIN_SENDER "${ENGINE}/../win32/OpenKey/OpenKey/OpenKey.cpp")
add_custom_command(OUTPUT "${SENDERS}"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/ExtractSenders.py" "${MAC_SENDER}" "${WIN_SENDER}" "${SENDERS}"
    DEPENDS "${MAC_SENDER}" "${WIN_SENDER}" ExtractSenders.py)
add_executable(sender_tests SenderTests.cpp EngineGlobals.cpp "${SENDERS}")
target_include_directories(sender_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sender_tests PRIVATE openkey_engine)
add_test(NAME senders COMMAND sender_tests)
if(WIN32)
    add_executable(update_installer_tests UpdateInstallerTests.cpp)
    add_test(NAME update_installer COMMAND update_installer_tests)
endif()
add_executable(engine_benchmark Benchmark.cpp EngineGlobals.cpp)
target_link_libraries(engine_benchmark PRIVATE openkey_engine)
